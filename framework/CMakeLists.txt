cmake_minimum_required(VERSION 3.14)
project(ve_qwen_inference C CXX)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../build/vendor/llama.cpp" CACHE PATH "Pinned llama.cpp source checkout")
add_subdirectory("${LLAMA_SOURCE_DIR}" llama)
add_executable(qwen-infer ../src/qwen_infer.cpp)
target_link_libraries(qwen-infer PRIVATE llama)
target_compile_features(qwen-infer PRIVATE cxx_std_17)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
    target_link_libraries(ggml PUBLIC stdc++fs)
endif()
add_executable(qwen-quant-probe ../tests/qwen_quant_probe.cpp)
target_link_libraries(qwen-quant-probe PRIVATE ggml ggml-cpu)
target_compile_features(qwen-quant-probe PRIVATE cxx_std_17)
if(GGML_OPENMP)
    find_package(OpenMP)
    if(OpenMP_CXX_FOUND)
        target_link_libraries(qwen-quant-probe PRIVATE OpenMP::OpenMP_CXX)
    endif()
endif()

target_include_directories(qwen-quant-probe PRIVATE "${LLAMA_SOURCE_DIR}/ggml/src")
# Float32 activation reference for the new block dequantization path.
add_library(qwen-float-math OBJECT ../src/qwen_nlc_hook.cpp ../src/ve_quant_kernels.c)
target_link_libraries(qwen-float-math PRIVATE ggml ggml-cpu)
target_include_directories(qwen-float-math PRIVATE "${LLAMA_SOURCE_DIR}/ggml/src" "${LLAMA_SOURCE_DIR}/ggml/src/ggml-cpu")
target_compile_features(qwen-float-math PRIVATE c_std_11 cxx_std_17)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(qwen-float-math PRIVATE -O3 -march=native)
endif()
add_executable(qwen-infer-cpu-float ../src/qwen_infer.cpp $<TARGET_OBJECTS:qwen-float-math>)
target_link_libraries(qwen-infer-cpu-float PRIVATE llama)
target_link_options(qwen-infer-cpu-float PRIVATE -Wl,--wrap=ggml_cpu_extra_compute_forward)
target_compile_features(qwen-infer-cpu-float PRIVATE cxx_std_17)
add_executable(check-qwen-nlc-cpu ../tests/check_qwen_nlc.cpp $<TARGET_OBJECTS:qwen-float-math>)
target_link_libraries(check-qwen-nlc-cpu PRIVATE ggml ggml-cpu)
target_link_options(check-qwen-nlc-cpu PRIVATE -Wl,--wrap=ggml_cpu_extra_compute_forward)
target_include_directories(check-qwen-nlc-cpu PRIVATE "${LLAMA_SOURCE_DIR}/ggml/src" "${LLAMA_SOURCE_DIR}/ggml/src/ggml-cpu")
if(OpenMP_CXX_FOUND)
    target_link_libraries(check-qwen-nlc-cpu PRIVATE OpenMP::OpenMP_CXX)
endif()
add_library(qwen-accurate-math OBJECT ../src/qwen_nlc_hook.cpp ../src/ve_quant_kernels.c)
target_link_libraries(qwen-accurate-math PRIVATE ggml ggml-cpu)
target_include_directories(qwen-accurate-math PRIVATE "${LLAMA_SOURCE_DIR}/ggml/src" "${LLAMA_SOURCE_DIR}/ggml/src/ggml-cpu")
target_compile_definitions(qwen-accurate-math PRIVATE QWEN_ACCUM_FP64)
target_compile_features(qwen-accurate-math PRIVATE c_std_11 cxx_std_17)
target_compile_options(qwen-accurate-math PRIVATE -O3 -fno-fast-math -ffp-contract=off)
add_executable(qwen-infer-cpu-accurate ../src/qwen_infer.cpp $<TARGET_OBJECTS:qwen-accurate-math>)
target_compile_definitions(qwen-infer-cpu-accurate PRIVATE QWEN_ACCUM_FP64)
target_link_libraries(qwen-infer-cpu-accurate PRIVATE llama)
target_compile_features(qwen-infer-cpu-accurate PRIVATE cxx_std_17)
target_link_options(qwen-infer-cpu-accurate PRIVATE -Wl,--wrap=ggml_cpu_extra_compute_forward -Wl,--wrap=ggml_vec_dot_f32)
add_executable(check-qwen-accurate-cpu ../tests/check_qwen_nlc.cpp $<TARGET_OBJECTS:qwen-accurate-math>)
target_compile_definitions(check-qwen-accurate-cpu PRIVATE QWEN_ACCUM_FP64)
target_link_libraries(check-qwen-accurate-cpu PRIVATE ggml ggml-cpu)
target_link_options(check-qwen-accurate-cpu PRIVATE -Wl,--wrap=ggml_cpu_extra_compute_forward -Wl,--wrap=ggml_vec_dot_f32)
target_include_directories(check-qwen-accurate-cpu PRIVATE "${LLAMA_SOURCE_DIR}/ggml/src" "${LLAMA_SOURCE_DIR}/ggml/src/ggml-cpu")
if(OpenMP_CXX_FOUND)
    target_link_libraries(check-qwen-accurate-cpu PRIVATE OpenMP::OpenMP_CXX)
endif()
add_executable(check-qwen-accurate-primitives-cpu ../tests/check_qwen_accurate.cpp $<TARGET_OBJECTS:qwen-accurate-math>)
target_link_libraries(check-qwen-accurate-primitives-cpu PRIVATE ggml ggml-cpu)
target_link_options(check-qwen-accurate-primitives-cpu PRIVATE -Wl,--wrap=ggml_cpu_extra_compute_forward -Wl,--wrap=ggml_vec_dot_f32)
target_include_directories(check-qwen-accurate-primitives-cpu PRIVATE "${LLAMA_SOURCE_DIR}/ggml/src" "${LLAMA_SOURCE_DIR}/ggml/src/ggml-cpu")
